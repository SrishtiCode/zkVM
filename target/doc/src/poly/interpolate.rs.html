1//! Turning `(x, y)` points into a polynomial that passes through them.
2//!
3//! Two ways to do it, matching the two situations that come up in a STARK
4//! pipeline:
5//!
6//! - [`lagrange_interpolate`] — works for *any* set of distinct `x`
7//!   coordinates, `O(n^2)`. Used e.g. when the domain isn't a nice
8//!   roots-of-unity set (arbitrary constraint points, small ad hoc checks).
9//! - [`fft_interpolate`] — requires the `x` coordinates to be exactly the
10//!   `n`-th roots of unity in order, but is `O(n log n)`. This is how an
11//!   execution trace's evaluation column gets turned back into a
12//!   polynomial in the real pipeline.
13
14use crate::fft;
15use crate::Polynomial;
16use field::Field;
17
18/// Lagrange interpolation: given `n` points with distinct `x` coordinates,
19/// returns the unique polynomial of degree `< n` passing through all of
20/// them.
21///
22/// `O(n^2)` field operations. Panics if two points share an `x`
23/// coordinate (the interpolant would be undefined / the system
24/// overdetermined).
25pub fn lagrange_interpolate<F: Field>(points: &[(F, F)]) -> Polynomial<F> {
26    assert!(!points.is_empty(), "cannot interpolate through zero points");
27
28    for i in 0..points.len() {
29        for j in (i + 1)..points.len() {
30            assert!(
31                points[i].0 != points[j].0,
32                "duplicate x-coordinate in interpolation points"
33            );
34        }
35    }
36
37    let mut result = Polynomial::zero();
38
39    for (i, &(xi, yi)) in points.iter().enumerate() {
40        // basis_i(x) = prod_{j != i} (x - x_j) / (x_i - x_j)
41        let mut numerator = Polynomial::new(vec![F::one()]);
42        let mut denominator = F::one();
43        for (j, &(xj, _)) in points.iter().enumerate() {
44            if i == j {
45                continue;
46            }
47            // multiply by the linear factor (x - x_j)
48            numerator = numerator.mul_naive(&Polynomial::new(vec![-xj, F::one()]));
49            denominator *= xi - xj;
50        }
51        let scale = yi
52            * denominator
53                .inverse()
54                .expect("distinct x-coordinates checked above");
55        result = result.add(&numerator.scale(scale));
56    }
57
58    result
59}

61/// Fast interpolation for the special case where the `x` coordinates are
62/// exactly `ω^0, ω^1, ..., ω^{n-1}` for `ω = F::root_of_unity(log2(n))`
63/// and `n` a power of two: `O(n log n)` via [`fft::ifft`], instead of
64/// `lagrange_interpolate`'s `O(n^2)`.
65///
66/// `evals[k]` must be the polynomial's value at `ω^k`.
67pub fn fft_interpolate<F: Field>(evals: &[F]) -> Polynomial<F> {
68    assert!(
69        evals.len().is_power_of_two(),
70        "fft_interpolate requires a power-of-two number of evaluations, got {}",
71        evals.len()
72    );
73    let mut coeffs = evals.to_vec();
74    fft::ifft(&mut coeffs);
75    Polynomial::new(coeffs)
76}

78#[cfg(test)]
79mod tests {
80    use super::*;
81    use field::ToyField;
82
83    fn tf(x: u64) -> ToyField {
84        ToyField::from_u64(x)
85    }
86
87    #[test]
88    fn lagrange_reproduces_known_polynomial() {
89        // p(x) = x^2 + 1 -> p(0)=1, p(1)=2, p(2)=5
90        let points = [(tf(0), tf(1)), (tf(1), tf(2)), (tf(2), tf(5))];
91        let p = lagrange_interpolate(&points);
92        for &(x, y) in &points {
93            assert_eq!(p.eval(x), y);
94        }
95        assert_eq!(p.coeffs(), &[tf(1), tf(0), tf(1)]);
96    }
97
98    #[test]
99    fn lagrange_handles_single_point_as_constant() {
100        let points = [(tf(3), tf(9))];
101        let p = lagrange_interpolate(&points);
102        assert_eq!(p.eval(tf(3)), tf(9));
103        assert_eq!(p.eval(tf(100)), tf(9)); // constant everywhere
104    }
105
106    #[test]
107    #[should_panic(expected = "duplicate x-coordinate")]
108    fn lagrange_rejects_duplicate_x() {
109        let points = [(tf(1), tf(2)), (tf(1), tf(5))];
110        let _ = lagrange_interpolate(&points);
111    }
112
113    #[test]
114    fn fft_interpolate_matches_lagrange_on_roots_of_unity() {
115        let log_n = 3u32;
116        let n = 1usize << log_n;
117        let omega = ToyField::root_of_unity(log_n);
118
119        // an arbitrary degree-<n polynomial to round-trip through
120        let original = Polynomial::new((1..=n as u64).map(tf).collect());
121
122        let mut x = ToyField::one();
123        let mut points = Vec::with_capacity(n);
124        let mut evals = Vec::with_capacity(n);
125        for _ in 0..n {
126            evals.push(original.eval(x));
127            points.push((x, original.eval(x)));
128            x *= omega;
129        }
130
131        let via_fft = fft_interpolate(&evals);
132        let via_lagrange = lagrange_interpolate(&points);
133
134        assert_eq!(via_fft, original);
135        assert_eq!(via_fft, via_lagrange);
136    }
137}
