1//! `Goldilocks` — the finite field `GF(2^64 - 2^32 + 1)`.
2//!
3//! This is a real field used by production STARK provers (Plonky2, Miden,
4//! RISC Zero's earlier circuits, ...), included so "real mode" in the
5//! visualizers can flip from [`crate::ToyField`] to this and show the exact
6//! same interpolation/FFT/FRI code operating at production scale.
7//!
8//! `p = 2^64 - 2^32 + 1` is chosen because:
9//! - it fits in a `u64`, so field elements are a single machine word,
10//! - `p - 1 = 2^32 * 3 * 5 * 17 * 257 * 65537`, giving a 2-adicity of 32 —
11//!   FFTs up to size `2^32` are supported, far more than any toy program
12//!   here will need.
13
14use crate::Field;
15use std::fmt;
16use std::ops::{Add, AddAssign, Mul, MulAssign, Neg, Sub, SubAssign};
17
18/// `2^64 - 2^32 + 1`.
19pub const MODULUS: u64 = 0xFFFF_FFFF_0000_0001;
20
21/// A primitive `2^32`-th root of unity mod `MODULUS`. This specific
22/// constant is the standard one used by Goldilocks-based provers
23/// (e.g. Plonky2); verified in this crate's tests to have order exactly
24/// `2^32`.
25const ROOT_OF_UNITY_32: u64 = 1_753_635_133_440_165_772;
26
27/// An element of `GF(2^64 - 2^32 + 1)`, stored canonically reduced
28/// (`0 <= value < MODULUS`).
29#[derive(Copy, Clone, PartialEq, Eq, Default, Hash)]
30pub struct Goldilocks(u64);
31
32impl Goldilocks {
33    /// Constructs an element from any `u64`, reducing mod `p` if needed.
34    pub fn new(value: u64) -> Self {
35        if value >= MODULUS {
36            Goldilocks(value - MODULUS)
37        } else {
38            Goldilocks(value)
39        }
40    }
41
42    /// The raw canonical representative in `0..MODULUS`.
43    pub fn value(&self) -> u64 {
44        self.0
45    }
46
47    /// Multiplies two canonical `u64`s mod `p` using a `u128` intermediate.
48    /// `p < 2^64` so the product fits comfortably in `u128` and a single
49    /// `% MODULUS` reduction is correct (no need for Goldilocks' fancier
50    /// bit-trick reduction at this stage — that's a performance concern
51    /// for later phases, not a correctness one for Phase 0).
52    fn mulmod(a: u64, b: u64) -> u64 {
53        ((a as u128 * b as u128) % MODULUS as u128) as u64
54    }
55
56    fn addmod(a: u64, b: u64) -> u64 {
57        // a, b < MODULUS < 2^64, so a + b < 2^65: widen to u128 to add
58        // without overflow, then reduce back down.
59        ((a as u128 + b as u128) % MODULUS as u128) as u64
60    }
61
62    fn submod(a: u64, b: u64) -> u64 {
63        if a >= b {
64            a - b
65        } else {
66            MODULUS - (b - a)
67        }
68    }
69}

83impl Add for Goldilocks {
84    type Output = Self;
85    fn add(self, rhs: Self) -> Self {
86        Goldilocks(Self::addmod(self.0, rhs.0))
87    }
88}

97impl Mul for Goldilocks {
98    type Output = Self;
99    fn mul(self, rhs: Self) -> Self {
100        Goldilocks(Self::mulmod(self.0, rhs.0))
101    }
102}

104impl Neg for Goldilocks {
105    type Output = Self;
106    fn neg(self) -> Self {
107        if self.0 == 0 {
108            self
109        } else {
110            Goldilocks(MODULUS - self.0)
111        }
112    }
113}

133impl Field for Goldilocks {
134    fn zero() -> Self {
135        Goldilocks(0)
136    }
137
138    fn one() -> Self {
139        Goldilocks(1)
140    }
141
142    fn from_u64(x: u64) -> Self {
143        Goldilocks::new(x)
144    }
145
146    fn modulus() -> u64 {
147        MODULUS
148    }
149
150    /// Fermat's little theorem: `a^(p-2) == a^-1` for `a != 0`, since
151    /// `GF(p)*` has order `p - 1`. Simpler and safer to get right than
152    /// extended-Euclid at this word size; performance can be revisited
153    /// later without changing the trait's contract.
154    fn inverse(&self) -> Option<Self> {
155        if self.0 == 0 {
156            None
157        } else {
158            Some(self.pow(MODULUS - 2))
159        }
160    }
161
162    const TWO_ADICITY: u32 = 32;
163
164    fn primitive_root_of_unity() -> Self {
165        Goldilocks(ROOT_OF_UNITY_32)
166    }
167}

169#[cfg(test)]
170mod tests {
171    use super::*;
172    use crate::field_laws;
173
174    #[test]
175    fn field_laws_hold() {
176        field_laws::run_all::<Goldilocks>();
177    }
178
179    #[test]
180    fn reduces_on_construction() {
181        assert_eq!(Goldilocks::new(MODULUS).value(), 0);
182        assert_eq!(Goldilocks::new(MODULUS + 5).value(), 5);
183    }
184
185    #[test]
186    fn add_near_overflow_boundary() {
187        let a = Goldilocks::new(MODULUS - 1);
188        let b = Goldilocks::new(2);
189        // (p - 1) + 2 = p + 1 ≡ 1 (mod p)
190        assert_eq!((a + b).value(), 1);
191    }
192
193    #[test]
194    fn root_of_unity_has_order_exactly_2_pow_32() {
195        let root = Goldilocks::primitive_root_of_unity();
196        assert_eq!(root.pow(1u64 << 32), Goldilocks::one());
197        assert_ne!(root.pow(1u64 << 31), Goldilocks::one());
198    }
199
200    #[test]
201    fn root_of_unity_domains_of_various_sizes() {
202        for log_n in [0u32, 1, 2, 3, 8, 16, 32] {
203            let n = 1u64 << log_n;
204            let root = Goldilocks::root_of_unity(log_n);
205            assert_eq!(root.pow(n), Goldilocks::one());
206            if n > 1 {
207                assert_ne!(root.pow(n / 2), Goldilocks::one());
208            }
209        }
210    }
211
212    #[test]
213    fn inverse_matches_extended_euclid_reference() {
214        // cross-check Fermat-based inverse against a from-scratch extended
215        // Euclid implementation, for a handful of values.
216        fn ext_euclid_inverse(a: u64, p: u64) -> u64 {
217            let (mut old_r, mut r) = (a as i128, p as i128);
218            let (mut old_s, mut s) = (1i128, 0i128);
219            while r != 0 {
220                let q = old_r / r;
221                let tmp_r = old_r - q * r;
222                old_r = r;
223                r = tmp_r;
224                let tmp_s = old_s - q * s;
225                old_s = s;
226                s = tmp_s;
227            }
228            (((old_s % p as i128) + p as i128) % p as i128) as u64
229        }
230
231        for x in [1u64, 2, 3, 12345, 999_999_999, MODULUS - 1] {
232            let a = Goldilocks::new(x);
233            let expected = ext_euclid_inverse(a.value(), MODULUS);
234            assert_eq!(a.inverse().unwrap().value(), expected);
235        }
236    }
