1//! `field` — finite field arithmetic (Phase 0).
2//!
3//! This crate defines the [`Field`] trait, the single abstraction the rest
4//! of the pipeline (poly, air, fri, ...) is built on top of, plus two
5//! concrete implementations:
6//!
7//! - [`toyfield::ToyField`] — arithmetic mod 97. Every element, every FFT
8//!   twiddle factor, every FRI fold is small enough to print and read by
9//!   eye. This is the field the visualizers use by default.
10//! - [`goldilocks::Goldilocks`] — the real 64-bit STARK-friendly field
11//!   `p = 2^64 - 2^32 + 1`. Same trait, same code paths, "real mode."
12//!
13//! Both fields have high **2-adicity**: `p - 1` is divisible by a large
14//! power of two, which is exactly what FFT-based polynomial arithmetic
15//! (interpolation, low-degree extension, FRI) needs, since it repeatedly
16//! splits a domain in half.
17
18pub mod goldilocks;
19pub mod toyfield;

27/// A finite field with a large 2-adic multiplicative subgroup.
28///
29/// Implementors get `+ - * neg` "for free" via the standard `std::ops`
30/// traits so field elements behave like any other numeric type in the rest
31/// of the codebase.
32pub trait Field:
33    Copy
34    + Clone
35    + Debug
36    + Default
37    + PartialEq
38    + Eq
39    + Add<Output = Self>
40    + Sub<Output = Self>
41    + Mul<Output = Self>
42    + Neg<Output = Self>
43    + AddAssign
44    + SubAssign
45    + MulAssign
46    + 'static
47{
48    /// The additive identity.
49    fn zero() -> Self;
50
51    /// The multiplicative identity.
52    fn one() -> Self;
53
54    /// `true` iff `self == Field::zero()`.
55    fn is_zero(&self) -> bool {
56        *self == Self::zero()
57    }
58
59    /// Embeds a `u64` into the field (reduced mod the field's characteristic).
60    fn from_u64(x: u64) -> Self;
61
62    /// The field's prime modulus, as a `u64`. Every field in this crate is
63    /// prime-order, so this fully determines the field.
64    fn modulus() -> u64;
65
66    /// Multiplicative inverse, or `None` for zero (which has none).
67    fn inverse(&self) -> Option<Self>;
68
69    /// `self^exp` via square-and-multiply.
70    fn pow(&self, exp: u64) -> Self {
71        let mut base = *self;
72        let mut exp = exp;
73        let mut acc = Self::one();
74        while exp > 0 {
75            if exp & 1 == 1 {
76                acc *= base;
77            }
78            base *= base;
79            exp >>= 1;
80        }
81        acc
82    }
83
84    /// `log2` of the largest power-of-two subgroup of the multiplicative
85    /// group, i.e. the largest `k` such that `2^k | (p - 1)`. FFTs of size
86    /// up to `2^TWO_ADICITY` are supported.
87    const TWO_ADICITY: u32;
88
89    /// A primitive `2^TWO_ADICITY`-th root of unity: `g` such that
90    /// `g^(2^TWO_ADICITY) == 1` and no smaller power of two exponent works.
91    fn primitive_root_of_unity() -> Self;
92
93    /// A primitive `2^log_n`-th root of unity, derived from
94    /// [`Self::primitive_root_of_unity`] by repeated squaring.
95    ///
96    /// This is the twiddle-factor generator every FFT/IFFT/LDE call in
97    /// `poly` uses to build its evaluation domain.
98    fn root_of_unity(log_n: u32) -> Self {
99        assert!(
100            log_n <= Self::TWO_ADICITY,
101            "requested domain of size 2^{log_n} exceeds field's 2-adicity 2^{}",
102            Self::TWO_ADICITY
103        );
104        let mut root = Self::primitive_root_of_unity();
105        for _ in 0..(Self::TWO_ADICITY - log_n) {
106            root *= root;
107        }
108        root
109    }
110}

112/// Shared property-based sanity checks, exercised against every `Field`
113/// impl in this crate's test modules. Kept generic so a new field added
114/// later gets the same coverage for free.
115#[cfg(test)]
116pub(crate) mod field_laws {
117    use super::Field;
118
119    pub fn check_additive_identity<F: Field>() {
120        let a = F::from_u64(1234567);
121        assert_eq!(a + F::zero(), a);
122        assert_eq!(F::zero() + a, a);
123    }
124
125    pub fn check_multiplicative_identity<F: Field>() {
126        let a = F::from_u64(7654321);
127        assert_eq!(a * F::one(), a);
128        assert_eq!(F::one() * a, a);
129    }
130
131    pub fn check_additive_inverse<F: Field>() {
132        let a = F::from_u64(42);
133        assert_eq!(a + (-a), F::zero());
134    }
135
136    pub fn check_multiplicative_inverse<F: Field>() {
137        for x in [1u64, 2, 3, 5, 97, 12345] {
138            let a = F::from_u64(x);
139            if a.is_zero() {
140                continue;
141            }
142            let inv = a.inverse().expect("nonzero element must be invertible");
143            assert_eq!(a * inv, F::one());
144        }
145        assert!(F::zero().inverse().is_none());
146    }
147
148    pub fn check_distributivity<F: Field>() {
149        let a = F::from_u64(3);
150        let b = F::from_u64(11);
151        let c = F::from_u64(19);
152        assert_eq!(a * (b + c), a * b + a * c);
153    }
154
155    pub fn check_root_of_unity<F: Field>() {
156        let log_n = 3u32.min(F::TWO_ADICITY);
157        let n = 1u64 << log_n;
158        let root = F::root_of_unity(log_n);
159        // root^n == 1
160        assert_eq!(root.pow(n), F::one());
161        // root^(n/2) != 1 (primitivity: no smaller power of two works)
162        if n > 1 {
163            assert_ne!(root.pow(n / 2), F::one());
164        }
165    }
166
167    pub fn run_all<F: Field>() {
168        check_additive_identity::<F>();
169        check_multiplicative_identity::<F>();
170        check_additive_inverse::<F>();
171        check_multiplicative_inverse::<F>();
172        check_distributivity::<F>();
173        check_root_of_unity::<F>();
174    }
175}
